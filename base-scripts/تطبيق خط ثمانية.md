/*

يطبّق خط ثمانية المحلي على النصوص المحددة، بما فيها النصوص المرتبطة بالأشكال.
يجب أولًا ضبط خط ثمانية بوصفه Local Font من إعدادات إضافة Excalidraw.

طريقة الاستخدام:
1. حدّد نصًا واحدًا أو عدة نصوص، أو حدّد الأشكال التي تحتوي نصوصًا.
2. شغّل السكربت من لوحة أدوات Excalidraw.
3. يحافظ السكربت على المحاذاة والحجم واللون، ويغيّر عائلة الخط فقط.

```javascript
*/

const selected = ea.getViewSelectedElements();

if (!selected || selected.length === 0) {
  new Notice("حدّد النصوص أو الأشكال التي تريد تطبيق خط ثمانية عليها");
  return;
}

const selectedIds = new Set(selected.map((element) => element.id));
const textElements = ea.getViewElements().filter(
  (element) =>
    element.type === "text" &&
    (selectedIds.has(element.id) ||
      (element.containerId && selectedIds.has(element.containerId))),
);

if (textElements.length === 0) {
  new Notice("لم أجد نصوصًا داخل التحديد الحالي");
  return;
}

textElements.forEach((element) => {
  element.fontFamily = 4;
});

ea.copyViewElementsToEAforEditing(textElements);
await ea.addElementsToView(false, false);
new Notice(`تم تطبيق خط ثمانية على ${textElements.length} عنصرًا نصيًا`);
